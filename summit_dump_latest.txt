import android.content.Context

/**
 * Summit's real-glass renderer.
 *
 * BlurView needs a View and a BlurTarget, unlike the old Drawable-only helper.
 * Keep creation/configuration here so UI code does not need to know BlurView's API.
 * During migration, surfaces not yet converted to real blur use the Drawable fallback
 * methods below, so their existing colors and opacity remain unchanged.
 */
object GlassSystem {
    // Centralized glass tuning. Lower alpha lets more of the page colors pass through.
    // Keep blur radius unchanged so the result remains frosted rather than clear acrylic.
    private const val BLURRED_OVERLAY_ALPHA_LIGHT = 38
    private const val BLURRED_OVERLAY_ALPHA_DARK = 34
    private const val FALLBACK_FILL_ALPHA_LIGHT = 152
    private const val FALLBACK_FILL_ALPHA_DARK = 142

    enum class Preset(val radiusDp: Int, val blurRadiusDp: Float) {
        ADDRESS_BAR(28, 18f),
        TOOLBAR_BUTTON(20, 16f),
        TOAST(22, 18f),
        SUGGESTIONS(22, 18f),
        FIND_BAR(28, 18f),
        MENU_CARD(26, 20f),
        ORB_BUTTON(20, 16f),
        ORB_SUB_BUTTON(16, 14f),
        TABS_DIALOG(0, 24f),
        TABS_BUTTON(18, 14f)
    }

    /**
     * Drawable fallback for surfaces that have not been converted to real blur yet.
     * Keep the previous Summit colors/opacity unchanged during the staged migration.
     */
    fun create(preset: Preset, density: Float, isNight: Boolean): GradientDrawable =
        create(preset.radiusDp, density, isNight)

    fun create(radiusDp: Int, density: Float, isNight: Boolean): GradientDrawable =
        GradientDrawable().apply {
            cornerRadius = radiusDp.coerceAtLeast(0) * density
            setColor(
                if (isNight) Color.argb(FALLBACK_FILL_ALPHA_DARK, 28, 28, 32)
                else Color.argb(FALLBACK_FILL_ALPHA_LIGHT, 255, 255, 255)
            )
            val strokeColor = if (isNight) Color.argb(60, 255, 255, 255)
                else Color.argb(40, 255, 255, 255)
            setStroke((density + 0.5f).toInt().coerceAtLeast(1), strokeColor)
        }

    fun createInputField(radiusDp: Int, density: Float, isNight: Boolean): GradientDrawable =
        GradientDrawable().apply {
            cornerRadius = radiusDp.coerceAtLeast(0) * density
            setColor(
                if (isNight) Color.argb(40, 255, 255, 255)
                else Color.argb(18, 0, 0, 0)
            )
        }

    /** Create a BlurView configured for a named glass surface. */
    fun createBlurView(
        context: Context,
        target: BlurTarget,
        preset: Preset,
        density: Float,
        isNight: Boolean
    ): BlurView {
        val shape = GradientDrawable().apply {
            cornerRadius = preset.radiusDp * density
            setColor(Color.TRANSPARENT)
            val strokeColor = if (isNight) {
                Color.argb(68, 255, 255, 255)
            } else {
                Color.argb(46, 255, 255, 255)
            }
            setStroke((density + 0.5f).toInt().coerceAtLeast(1), strokeColor)
        }

        val overlayColor = if (isNight) {
            Color.argb(BLURRED_OVERLAY_ALPHA_DARK, 28, 28, 32)
        } else {
            Color.argb(BLURRED_OVERLAY_ALPHA_LIGHT, 255, 255, 255)
        }

        return BlurView(context).apply {
            background = shape
            outlineProvider = ViewOutlineProvider.BACKGROUND
            clipToOutline = true
            setupWith(target)
                .setBlurRadius(preset.blurRadiusDp * density)
                .setOverlayColor(overlayColor)
        }
    }
}

class PathIcon(pathData: String, color: Int) : Drawable() {
    private val path: Path? = try {
        PathParser.createPathFromPathData(pathData)
    } catch (e: Exception) {
        null
    }
    private val paint = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        this.color = color
        style = Paint.Style.FILL
    }
    override fun draw(canvas: Canvas) {
        val p = path ?: return
        val b = bounds
        if (b.width() <= 0 || b.height() <= 0) return
        val sc = b.width() / 24f
        canvas.save()
        canvas.translate(b.left.toFloat(), b.top.toFloat())
        canvas.scale(sc, sc)
        canvas.drawPath(p, paint)
        canvas.restore()
    }
    override fun setAlpha(alpha: Int) { paint.alpha = alpha }
    override fun setColorFilter(colorFilter: ColorFilter?) { paint.colorFilter = colorFilter }
    @Suppress("OVERRIDE_DEPRECATION")
    override fun getOpacity(): Int = PixelFormat.TRANSLUCENT
}

private fun handleSpecialUrl(url: String, fallbackView: WebView = web): Boolean {
    val uri = try { Uri.parse(url) } catch (_: Exception) { return false }
    val scheme = uri.scheme?.lowercase(Locale.ROOT) ?: return false
    if (scheme in setOf("http", "https", "about", "javascript", "data", "blob", "file")) {
    return false
    }
    return try {
    val intent = if (scheme == "intent") {
    Intent.parseUri(url, Intent.URI_INTENT_SCHEME)
    } else {
    Intent(Intent.ACTION_VIEW, uri)
    }
    intent.component = null
    intent.addCategory(Intent.CATEGORY_BROWSABLE)
    val fallback = intent.getStringExtra("browser_fallback_url")?.let {
    val fallbackUri = runCatching { Uri.parse(it) }.getOrNull()
    if (fallbackUri?.scheme.equals("https", ignoreCase = true) ||
    fallbackUri?.scheme.equals("http", ignoreCase = true)) it else null
    }
    when {
    intent.resolveActivity(packageManager) != null -> {
    startActivity(intent)
    true
    }
    !fallback.isNullOrBlank() -> {
    fallbackView.loadUrl(fallback)
    true
    }
    else -> {
    toast("Không tìm thấy ứng dụng để mở liên kết này")
    true
    }
    }
    } catch (_: Exception) {
    toast("Không thể mở liên kết bên ngoài")
    true
    }
    }
private fun showFullscreenView(
view: View,
callback: WebChromeClient.CustomViewCallback,
requestedOrientationValue: Int? = null
) {
if (!::fullscreenContainer.isInitialized || customVideoView != null) {
callback.onCustomViewHidden()
return
}
customVideoView = view
customVideoCallback = callback
previousRequestedOrientation = requestedOrientation
requestedOrientationValue?.let {
if (it != ActivityInfo.SCREEN_ORIENTATION_UNSPECIFIED) requestedOrientation = it
}
(view.parent as? ViewGroup)?.removeView(view)
fullscreenContainer.removeAllViews()
fullscreenContainer.addView(view, FrameLayout.LayoutParams(-1, -1))
fullscreenContainer.visibility = View.VISIBLE
fullscreenContainer.bringToFront()
orbController?.setTemporarilyHidden(true)
WindowInsetsControllerCompat(window, root).hide(WindowInsetsCompat.Type.systemBars())
}

private fun exitFullscreenView() {
val view = customVideoView ?: return
if (::fullscreenContainer.isInitialized) {
fullscreenContainer.removeView(view)
fullscreenContainer.visibility = View.GONE
}
customVideoView = null
val callback = customVideoCallback
customVideoCallback = null
previousRequestedOrientation?.let { requestedOrientation = it }
previousRequestedOrientation = null
if (!isActivityDestroyed && !isFinishing && !isDestroyed && ::root.isInitialized) {
WindowInsetsControllerCompat(window, root).show(WindowInsetsCompat.Type.systemBars())
}
syncOrbMenuVisibility()
callback?.onCustomViewHidden()
}

    // Nhấn co lại + rung nhẹ khi nhả tay trong vùng nút
    @SuppressLint("ClickableViewAccessibility")
    private fun press(v: View, scale: Float) {
        v.setOnTouchListener { view, ev ->
            when (ev.actionMasked) {
                MotionEvent.ACTION_DOWN -> {
                    view.animate().scaleX(scale).scaleY(scale).setDuration(90).start()
                }
                MotionEvent.ACTION_UP -> {
                    view.animate().scaleX(1f).scaleY(1f).setDuration(180)
                        .setInterpolator(OvershootInterpolator(2f)).start()
                    if (ev.x >= 0f && ev.x <= view.width && ev.y >= 0f && ev.y <= view.height) {
                        view.performHapticFeedback(HapticFeedbackConstants.CONTEXT_CLICK)
                    }
                }
                MotionEvent.ACTION_CANCEL -> {
                    view.animate().scaleX(1f).scaleY(1f).setDuration(180).start()
                }
            }
            false
        }
    }

    // Chừa chỗ đầu/cuối trang cho các thanh và chạy script xử lý thanh menu cố định
    private fun injectInsets() {
        val edge = store.edgeMode
        val topCss = if (edge) (insetTop / d).toInt() else 0
        val bottomCss = (insetBottom / d).toInt() + 64
        val rule = (if (edge) "padding-top:" + topCss + "px!important;" else "") +
            "padding-bottom:" + bottomCss + "px!important;box-sizing:border-box!important"
        val js = "(function(){var s=document.getElementById('__edge');" +
            "if(!s){s=document.createElement('style');s.id='__edge';" +
            "(document.head||document.documentElement).appendChild(s);}" +
            "s.textContent='html{" + rule + "}';})();"
        web.evaluateJavascript(js, null)
        web.evaluateJavascript(
            EDGE_JS.replace("%INSET%", (insetTop / d).toInt().toString())
                .replace("%MODE%", if (edge) "1" else "0"),
            null
        )
    }

    private fun scheduleSuggest(q: String) {
        suggestRunnable?.let { handler.removeCallbacks(it) }
        val text = q.trim()
        if (text.isEmpty()) {
            hideSuggestions()
            return
        }
        val r = Runnable { fetchSuggestions(text) }
        suggestRunnable = r
        handler.postDelayed(r, 150)
    }

    private fun fetchSuggestions(q: String) {
        suggestToken++
        val token = suggestToken
        val engine = store.searchEngine
        Thread {
            val result = try {
                requestSuggestions(engine, q)
            } catch (e: Exception) {
                emptyList<String>()
            }
            runOnUiThread {
                if (!isActivityDestroyed && token == suggestToken && urlBar.hasFocus()) showSuggestions(result)
            }
        }.start()
    }

    private fun requestSuggestions(engine: Int, q: String): List<String> {
        val enc = URLEncoder.encode(q, "UTF-8")
        val link = when (engine) {
            1 -> "https://api.bing.com/osjson.aspx?query=" + enc
            2 -> "https://duckduckgo.com/ac/?type=list&q=" + enc
            else -> "https://suggestqueries.google.com/complete/search?client=firefox&q=" + enc
        }
        val conn = URL(link).openConnection() as HttpURLConnection
        conn.connectTimeout = 3000
        conn.readTimeout = 3000
        conn.setRequestProperty("User-Agent", "Mozilla/5.0")
        try {
            val body = conn.inputStream.bufferedReader(Charsets.UTF_8).use { it.readText() }
            val arr = JSONArray(body).getJSONArray(1)
            val out = ArrayList<String>()
            for (i in 0 until minOf(arr.length(), 6)) out.add(arr.getString(i))
            return out
        } finally {
            conn.disconnect()
        }
    }

        if (items.isEmpty()) {

            row.orientation = LinearLayout.HORIZONTAL

            row.setPadding(dp(18), 0, dp(18), 0)

            t.textSize = 15f
            t.setTextColor(fg)
            t.setSingleLine()

            row.addView(t, LinearLayout.LayoutParams(0, ViewGroup.LayoutParams.WRAP_CONTENT, 1f))

            }
            press(row, 0.98f)
            suggestBox.addView(row, LinearLayout.LayoutParams(ViewGroup.LayoutParams.MATCH_PARENT, dp(44)))
        }
        suggestBlurView.visibility = View.VISIBLE
        suggestBox.visibility = View.VISIBLE
    }

    private fun searchBase(): String = when (store.searchEngine) {
        1 -> "https://www.bing.com/search?q="
        2 -> "https://duckduckgo.com/?q="
        else -> "https://www.google.com/search?q="

    private fun toUrl(input: String): String {
        val t = input.trim()
        return when {
            t.startsWith("http://") || t.startsWith("https://") -> t
            t.contains(".") && !t.contains(" ") -> "https://" + t
            else -> searchBase() + Uri.encode(t)
        }
    }

    private fun load(input: String) {
        if (input.isBlank()) return
        web.loadUrl(toUrl(input))
    }

        }

    // Thay nội dung khung menu bằng một trang khác, có hiệu ứng đổi kích thước
    private fun setMenuPage(page: View, widthDp: Int, animate: Boolean) {
        val layer = menuLayer ?: return
        val card = menuCard ?: return
        val w = minOf(dp(widthDp), resources.displayMetrics.widthPixels - dp(28) - insetLeft - insetRight)
        if (animate) {
            val tr = AutoTransition()
            tr.duration = 260
            tr.interpolator = DecelerateInterpolator()
            TransitionManager.beginDelayedTransition(layer, tr)
        }
        // Giữ lớp kính ở child đầu tiên, chỉ thay nội dung trang menu.
        // Reset kích thước kính để nội dung mới tự quyết định kích thước card.
        if (card.childCount > 0) {
            val glassParams = card.getChildAt(0).layoutParams as? FrameLayout.LayoutParams
            if (glassParams != null && (glassParams.width != 0 || glassParams.height != 0)) {
                glassParams.width = 0
                glassParams.height = 0
                card.getChildAt(0).layoutParams = glassParams
            }
        }
        if (card.childCount > 1) {
            card.removeViews(1, card.childCount - 1)
        }
        card.addView(page, FrameLayout.LayoutParams(w, ViewGroup.LayoutParams.WRAP_CONTENT))
        syncMenuGlassBounds(card)
        WindowCompat.getInsetsController(window, layer).hide(WindowInsetsCompat.Type.ime())
    }

    /** Size the BlurView after the wrap-content menu card has been measured. */
    private fun syncMenuGlassBounds(card: FrameLayout) {
        val observer = card.viewTreeObserver
        observer.addOnGlobalLayoutListener(object : android.view.ViewTreeObserver.OnGlobalLayoutListener {
            override fun onGlobalLayout() {
                if (menuCard !== card || card.parent == null || card.childCount == 0) {
                    if (observer.isAlive) observer.removeOnGlobalLayoutListener(this)
                    return
                }

                val width = card.width
                val height = card.height
                if (width <= 0 || height <= 0) return

                val glass = card.getChildAt(0)
                val params = glass.layoutParams as? FrameLayout.LayoutParams ?: return
                if (params.width == width && params.height == height) {
                    if (observer.isAlive) observer.removeOnGlobalLayoutListener(this)
                    return
                }

                params.width = width
                params.height = height
                glass.layoutParams = params
                // Remove this listener after the first successful size sync.
                // A new listener is registered whenever setMenuPage changes content.
                if (observer.isAlive) observer.removeOnGlobalLayoutListener(this)
            }
        })
    }

    private fun showMainPage(animate: Boolean = true) {

        fun cell(row: LinearLayout, label: String, path: String, action: () -> Unit) {

            c.gravity = Gravity.CENTER
            c.setBackgroundResource(rp)

                closeMenu()

            press(c, 0.92f)
            row.addView(c, LinearLayout.LayoutParams(0, ViewGroup.LayoutParams.MATCH_PARENT, 1f))
        }

        // closes = true: đóng menu rồi thực hiện; false: mở trang mới ngay trong khung menu
        fun item(label: String, path: String?, closes: Boolean, action: () -> Unit) {
            val r = LinearLayout(this)
            r.orientation = LinearLayout.HORIZONTAL
            r.gravity = Gravity.CENTER_VERTICAL

    // Rút gọn link: chỉ hiện tên miền + đường dẫn, bỏ phần tham số dài
    private fun shortUrl(url: String): String {
        return try {
            val u = Uri.parse(url)
            val host = u.host?.removePrefix("www.") ?: return url
            val path = u.path ?: ""
            if (path.length > 1) host + path else host
        } catch (e: Exception) {
            url
        }
    }

    private fun showList(title: String, key: String) {
        val p = makePanel(title, "Chạm để mở, giữ lâu để xóa", 340) { showMainPage() }
        val rp = ripple()

        fun fill() {
            p.body.removeAllViews()

            if (current.isEmpty()) {
                val empty = TextView(this)
                empty.text = "Chưa có gì"
                empty.textSize = 15f
                empty.setTextColor(sub)
                empty.setPadding(dp(22), dp(16), dp(22), dp(16))
                p.body.addView(empty)
                return
            }
            for (entry in current) {
                val row = LinearLayout(this)
                row.orientation = LinearLayout.VERTICAL
                row.setPadding(dp(22), dp(10), dp(22), dp(10))
                row.setBackgroundResource(rp)

                val t = TextView(this)

                t.textSize = 15f
                t.setTextColor(fg)
                t.setSingleLine()
                t.ellipsize = TextUtils.TruncateAt.END
                row.addView(t)

                u.text = shortUrl(entry.url)
                u.textSize = 12f
                u.setTextColor(sub)
                u.setSingleLine()
                u.ellipsize = TextUtils.TruncateAt.END
                row.addView(u)

                row.setOnClickListener {

                }

                }

            }
        }

        val left: TextView? = if (key == "history") {

            }

    override fun onDestroy() {
        isActivityDestroyed = true
        denyPendingWebPermission()
        pendingGeoCallback?.invoke(pendingGeoOrigin.orEmpty(), false, false)
        pendingGeoCallback = null
        pendingGeoOrigin = null
        popupDialog?.dismiss()
        if (customVideoView != null) exitFullscreenView()
        suggestToken++
        // === CLEANUP DE TRANH MEMORY LEAK ===
        // 1. Gỡ JS interface (tránh inner class Bridge giữ reference)
        web.removeJavascriptInterface("EdgeBridge")
        
        // 2. Gỡ listeners khỏi WebView
        web.webViewClient = WebViewClient()
        web.webChromeClient = WebChromeClient()
        web.setDownloadListener(null)
        web.setFindListener(null)
        web.setOnScrollChangeListener(null as View.OnScrollChangeListener?)
        
        // 3. Hủy tất cả Runnable đang chờ trong Handler
        handler.removeCallbacksAndMessages(null)
        suggestRunnable?.let { handler.removeCallbacks(it) }
        suggestRunnable = null
        
        // 4. Hủy callbacks
        pendingFileCallback?.onReceiveValue(null)
        pendingFileCallback = null
        
        // 5. Cleanup Orb Menu Controller (gỡ PreDrawListener)
        orbController?.destroy()
        orbController = null
        
        // 6. Destroy WebView (phải làm CUỐI CÙNG)
        // Giai phong thumbnail cua cac tab
        tabManager.getAll().forEach { it.thumbnail?.recycle(); it.thumbnail = null }
        web.stopLoading()
        // Do not clear persistent WebView cache/history during ordinary Activity teardown.
        // The explicit "Xóa dữ liệu duyệt web" action remains responsible for data clearing.
        web.loadUrl("about:blank")
        web.onPause()
        web.removeAllViews()
        web.destroyDrawingCache()
        web.destroy()
        
        super.onDestroy()
    }

    private class BadgeDrawable(
        private val count: Int,
        private val bgColor: Int,
        private val textColor: Int,
        private val d: Float
    ) : android.graphics.drawable.Drawable() {
        private val bgPaint = android.graphics.Paint(android.graphics.Paint.ANTI_ALIAS_FLAG).apply { color = bgColor }
        private val textPaint = android.graphics.Paint(android.graphics.Paint.ANTI_ALIAS_FLAG).apply {
            color = textColor
            textAlign = android.graphics.Paint.Align.CENTER
            textSize = 10 * d
            typeface = android.graphics.Typeface.DEFAULT_BOLD
        }
        override fun draw(canvas: android.graphics.Canvas) {
            val b = bounds
            if (b.width() <= 0 || b.height() <= 0) return
            val cx = b.exactCenterX()
            val cy = b.exactCenterY()
            val r = minOf(b.width(), b.height()) / 2f
            canvas.drawCircle(cx, cy, r, bgPaint)
            val text = if (count > 99) "99+" else count.toString()
            val y = cy - (textPaint.descent() + textPaint.ascent()) / 2f
            canvas.drawText(text, cx, y, textPaint)
        }
        override fun setAlpha(alpha: Int) {}
        override fun setColorFilter(cf: android.graphics.ColorFilter?) {}
        @Suppress("OVERRIDE_DEPRECATION")
        override fun getOpacity(): Int = android.graphics.PixelFormat.TRANSLUCENT
    }

    private fun updateTabBadge() {
        val count = tabManager.count()
        val base = PathIcon(Icons.TABS, fg)
        if (count <= 1) {
            tabBtn.setImageDrawable(base)
            return
        }
        val badgeColor = if (isNight) Color.rgb(50, 140, 255) else Color.rgb(10, 132, 255)
        val textColor = if (isNight) Color.BLACK else Color.WHITE
        val layer = android.graphics.drawable.LayerDrawable(arrayOf(
            base,
            BadgeDrawable(count, badgeColor, textColor, d)
        ))
        layer.setLayerInset(1, dp(14), 0, 0, dp(14))
        tabBtn.setImageDrawable(layer)
    }

    private fun openTabSwitcher() {
        val popup = Dialog(this, android.R.style.Theme_Translucent_NoTitleBar)
        popup.requestWindowFeature(Window.FEATURE_NO_TITLE)

        val container = FrameLayout(this)
        // Nen kinh mo Liquid Glass (blur tu noi dung web phia sau)

        ).apply {
            isClickable = false
            isFocusable = false
        }

        root.orientation = LinearLayout.VERTICAL
        root.setPadding(dp(12), dp(24), dp(12), dp(12))
        root.setOnApplyWindowInsetsListener { view, insets ->
            val topInset = insets.systemWindowInsetTop
            view.setPadding(dp(12), topInset + dp(8), dp(12), dp(12))
            insets
        }
        // Root trong suot de BlurView phia sau hien thi (chi tint nhe de doc chu de)
        root.setBackgroundColor(if (isNight) Color.argb(100, 24, 24, 28) else Color.argb(90, 245, 245, 250))
        root.isClickable = true

        val header = LinearLayout(this)
        header.orientation = LinearLayout.HORIZONTAL
        header.gravity = Gravity.CENTER_VERTICAL
        header.setPadding(dp(8), dp(8), dp(8), dp(16))

        val heading = LinearLayout(this)
        heading.orientation = LinearLayout.VERTICAL

        val title = TextView(this)
        title.text = "Tab đang mở"
        title.textSize = 28f
        title.setTypeface(null, Typeface.BOLD)
        title.setTextColor(fg)
        heading.addView(title)

        val count = TextView(this)
        count.text = "${tabManager.count()} tab"
        count.textSize = 13f
        count.setTextColor(sub)
        heading.addView(count)

        header.addView(heading, LinearLayout.LayoutParams(0, ViewGroup.LayoutParams.WRAP_CONTENT, 1f))

        val doneFrame = createBlurButton(
            text = "Xong",
            paddingH = 18,
            paddingV = 6,
            pressScale = 0.92f
        ) { popup.dismiss() }
        header.addView(doneFrame)
        root.addView(header)

        val scroll = ScrollView(this)
        scroll.isFillViewport = true
        val grid = LinearLayout(this)
        grid.orientation = LinearLayout.VERTICAL

        val tabs = tabManager.getAll()
        for (chunk in tabs.chunked(2)) {
            val row = LinearLayout(this)
            row.orientation = LinearLayout.HORIZONTAL
            row.gravity = Gravity.TOP

            for (tab in chunk) {
                val card = LinearLayout(this)
                card.orientation = LinearLayout.VERTICAL
                card.setPadding(dp(7), dp(7), dp(7), dp(9))
                card.background = android.graphics.drawable.GradientDrawable().apply {
                    shape = android.graphics.drawable.GradientDrawable.RECTANGLE
                    cornerRadius = dp(18).toFloat()
                    setColor(if (isNight) Color.argb(180, 42, 42, 48) else Color.argb(200, 255, 255, 255))
                    setStroke(
                        dp(if (tab.isActive) 2 else 1),
                        if (tab.isActive) accent else
                            if (isNight) Color.argb(120, 120, 120, 130) else Color.argb(120, 225, 225, 232)
                    )
                }

                val preview = FrameLayout(this)
                preview.background = android.graphics.drawable.GradientDrawable(
                    android.graphics.drawable.GradientDrawable.Orientation.TL_BR,
                    if (isNight)
                        intArrayOf(Color.argb(180, 63, 67, 82), Color.argb(180, 35, 39, 52))
                    else
                        intArrayOf(Color.argb(200, 225, 232, 246), Color.argb(200, 190, 204, 231))
                ).apply {
                    cornerRadius = dp(13).toFloat()
                }

                // Hien thumbnail neu co, nguoc lai hien chu cai dau voi mau sac rieng


                if (tab.thumbnail != null) {


                    val iv = ImageView(this)


                    iv.setImageBitmap(tab.thumbnail)


                    iv.scaleType = ImageView.ScaleType.CENTER_CROP


                    iv.setBackgroundColor(Color.WHITE)


                    preview.addView(iv, FrameLayout.LayoutParams(


                        ViewGroup.LayoutParams.MATCH_PARENT,


                        ViewGroup.LayoutParams.MATCH_PARENT


                    ))


                } else {


                    val domain = try { Uri.parse(tab.url).host?.removePrefix("www.") ?: "S" } catch (_: Exception) { "S" }


                    val firstChar = domain.take(1).uppercase()


                    val hash = tab.url.hashCode()


                    val hue = (hash and 0xFF) % 360


                    val sat = if (isNight) 0.5f else 0.6f


                    val lum = if (isNight) 0.6f else 0.45f


                    val color = android.graphics.Color.HSVToColor(floatArrayOf(hue.toFloat(), sat, lum))


                    val mark = TextView(this)


                    mark.text = firstChar


                    mark.textSize = 46f


                    mark.setTypeface(null, Typeface.BOLD)


                    mark.setTextColor(color)


                    mark.gravity = Gravity.CENTER


                    preview.addView(mark, FrameLayout.LayoutParams(


                        ViewGroup.LayoutParams.MATCH_PARENT,


                        ViewGroup.LayoutParams.MATCH_PARENT


                    ))


                }

                if (!tab.isActive) {
                    val close = TextView(this)
                    close.text = "×"
                    close.textSize = 23f
                    close.gravity = Gravity.CENTER

class RingIcon(color: Int) : Drawable() {
    private val paint = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        this.color = color
        style = Paint.Style.STROKE
        strokeCap = Paint.Cap.ROUND
    }

    override fun draw(canvas: Canvas) {
        val b = bounds
        if (b.width() <= 0 || b.height() <= 0) return

        val strokeWidth = b.width() / 18f
        paint.strokeWidth = strokeWidth
        val cx = b.exactCenterX()
        val cy = b.exactCenterY()
        val radius = minOf(b.width(), b.height()) / 2f - strokeWidth
        canvas.drawCircle(cx, cy, radius, paint)
    }

    override fun setAlpha(alpha: Int) { paint.alpha = alpha }
    override fun setColorFilter(colorFilter: ColorFilter?) { paint.colorFilter = colorFilter }

    @Suppress("OVERRIDE_DEPRECATION")
    override fun getOpacity(): Int = PixelFormat.TRANSLUCENT
}

class OrbMenuController(
    private val rootLayout: FrameLayout,
    private val blurTarget: BlurTarget,
    private val dp: (Int) -> Int,
    private val d: Float,
    private val isNight: Boolean,
    private val BOTTOM_BAR_HEIGHT_DP: Int,
    private val onBack: () -> Unit,
    private val onForward: () -> Unit,
    private val onHome: () -> Unit
) {
    private val iconColor = if (isNight) Color.WHITE else Color.rgb(28, 28, 32)

    private var toolbarAvailable = true
    private var temporarilyHidden = false
    private var isExpanded = false
    private var anchorView: View? = null
    private var preDrawListener: ViewTreeObserver.OnPreDrawListener? = null

    private fun createGlass(preset: GlassSystem.Preset): View =
        GlassSystem.createBlurView(rootLayout.context, blurTarget, preset, d, isNight)

    private fun createIcon(path: String?, ring: Boolean = false): ImageView =
        ImageView(rootLayout.context).apply {
            setImageDrawable(if (ring) RingIcon(iconColor) else PathIcon(path ?: "", iconColor))
            scaleType = ImageView.ScaleType.FIT_CENTER
            setPadding(dp(7), dp(7), dp(7), dp(7))
            importantForAccessibility = View.IMPORTANT_FOR_ACCESSIBILITY_NO
        }

    private val mainBtn = FrameLayout(rootLayout.context).apply {
        isClickable = true
        isFocusable = true
        clipChildren = true
        clipToPadding = true

        addView(createGlass(GlassSystem.Preset.ORB_BUTTON), FrameLayout.LayoutParams(-1, -1))
        addView(
            createIcon(null, ring = true).apply { setPadding(dp(8), dp(8), dp(8), dp(8)) },
            FrameLayout.LayoutParams(-1, -1)
        )
        elevation = dp(3).toFloat()
    }

    private val subBtns = mutableListOf<FrameLayout>()

    init {
        rootLayout.addView(
            mainBtn,
            FrameLayout.LayoutParams(dp(BOTTOM_BAR_HEIGHT_DP), dp(BOTTOM_BAR_HEIGHT_DP)).apply {
                gravity = Gravity.TOP or Gravity.START
                leftMargin = dp(16)
                topMargin = dp(16)
            }
        )
        mainBtn.setOnClickListener { toggle() }

        val icons = listOf(Icons.ARROW_BACK, Icons.ARROW_FORWARD, Icons.HOME)
        val labels = listOf("Quay lại", "Tiến tới", "Trang chủ")
        val actions = listOf(onBack, onForward, onHome)

        for (i in icons.indices) {
            val btn = FrameLayout(rootLayout.context).apply {
                isClickable = true
                isFocusable = true
                contentDescription = labels[i]
                clipChildren = true
                clipToPadding = true
                addView(createGlass(GlassSystem.Preset.ORB_SUB_BUTTON), FrameLayout.LayoutParams(-1, -1))
                addView(createIcon(icons[i]), FrameLayout.LayoutParams(-1, -1))
                setOnClickListener {
                    if (isAvailable()) {
                        actions[i]()
                        collapse()
                    }
                }
                visibility = View.GONE
                scaleX = 0f
                scaleY = 0f
                alpha = 0f
                elevation = dp(3).toFloat()
            }
            rootLayout.addView(
                btn,
                FrameLayout.LayoutParams(dp(32), dp(32)).apply {
                    gravity = Gravity.TOP or Gravity.START
                    leftMargin = dp(20)
                    topMargin = dp(20)
                }
            )
            subBtns.add(btn)
        }
        applyAvailability()
    }

    fun alignWith(backBtn: View) {
        anchorView?.viewTreeObserver?.let { oldObserver ->
            preDrawListener?.let { if (oldObserver.isAlive) oldObserver.removeOnPreDrawListener(it) }
        }
        anchorView = backBtn
        updatePosition()
        val listener = ViewTreeObserver.OnPreDrawListener {
            updatePosition()
            true
        }
        preDrawListener = listener
        backBtn.viewTreeObserver.addOnPreDrawListener(listener)
    }

    /** Sync visibility with the real toolbar state (including auto-hide). */
    fun setToolbarAvailable(available: Boolean) {
        if (toolbarAvailable == available) {
            applyAvailability()
            return
        }
        toolbarAvailable = available
        if (!available) collapseImmediately()
        applyAvailability()
        updatePosition()
    }

    /**
     * Remove OrbMenu views from the root while FindBar/IME overlays are active.
     * This is stronger than visibility alone and prevents stale/animated views
     * from remaining above the FindBar or receiving touches.
     */
    fun setTemporarilyHidden(hidden: Boolean) {
        temporarilyHidden = hidden
        if (hidden) {
            collapseImmediately()
            mainBtn.animate().cancel()
            mainBtn.alpha = 0f
            mainBtn.isClickable = false
            mainBtn.isEnabled = false
            mainBtn.isFocusable = false
            mainBtn.visibility = View.GONE
            subBtns.forEach { btn ->
                btn.animate().cancel()
                btn.alpha = 0f
                btn.scaleX = 0f
                btn.scaleY = 0f
                btn.isClickable = false
                btn.isEnabled = false
                btn.visibility = View.GONE
            }
            // Detaching guarantees the menu cannot overlay FindBar or intercept touches.
            (mainBtn.parent as? ViewGroup)?.removeView(mainBtn)
            subBtns.forEach { btn -> (btn.parent as? ViewGroup)?.removeView(btn) }
        } else {
            if (mainBtn.parent == null) {
                rootLayout.addView(mainBtn, mainBtn.layoutParams ?: FrameLayout.LayoutParams(
                    dp(BOTTOM_BAR_HEIGHT_DP), dp(BOTTOM_BAR_HEIGHT_DP)
                ).apply { gravity = Gravity.TOP or Gravity.START; leftMargin = dp(16); topMargin = dp(16) })
            }
            subBtns.forEachIndexed { index, btn ->
                if (btn.parent == null) {
                    val params = btn.layoutParams ?: FrameLayout.LayoutParams(dp(32), dp(32)).apply {
                        gravity = Gravity.TOP or Gravity.START
                        leftMargin = dp(20)
                        topMargin = dp(20)
                    }
                    rootLayout.addView(btn, params)
                }
            }
            mainBtn.alpha = 1f
            applyAvailability()
            updatePosition()
        }
    }

    private fun isAvailable(): Boolean = toolbarAvailable && !temporarilyHidden

    private fun applyAvailability() {
        val available = isAvailable()
        mainBtn.visibility = if (available) View.VISIBLE else View.GONE
        mainBtn.isClickable = available
        mainBtn.isEnabled = available
        mainBtn.isFocusable = available
        if (!available) {
            subBtns.forEach {
                it.animate().cancel()
                it.visibility = View.GONE
                it.isClickable = false
                it.isEnabled = false
                it.alpha = 0f
                it.scaleX = 0f
                it.scaleY = 0f
                it.translationX = 0f
                it.translationY = 0f
            }
        } else {
            subBtns.forEach { it.isClickable = isExpanded; it.isEnabled = isExpanded }
        }
    }

    private fun updatePosition() {
        val anchor = anchorView ?: return
        if (anchor.width <= 0) {
            applyAvailability()
            return
        }

        val loc = IntArray(2)
        val rootLoc = IntArray(2)
        anchor.getLocationInWindow(loc)
        rootLayout.getLocationInWindow(rootLoc)
        val x = loc[0] - rootLoc[0]
        val y = loc[1] - rootLoc[1]

        val mainParams = mainBtn.layoutParams as FrameLayout.LayoutParams
        if (mainParams.leftMargin != x || mainParams.topMargin != y ||
            mainParams.gravity != (Gravity.TOP or Gravity.START)
        ) {
            mainParams.leftMargin = x
            mainParams.topMargin = y
            mainParams.gravity = Gravity.TOP or Gravity.START
            mainBtn.layoutParams = mainParams
        }

        subBtns.forEach { btn ->
            val params = btn.layoutParams as FrameLayout.LayoutParams
            val newX = x + dp(4)
            val newY = y + dp(4)
            if (params.leftMargin != newX || params.topMargin != newY ||
                params.gravity != (Gravity.TOP or Gravity.START)
            ) {
                params.leftMargin = newX
                params.topMargin = newY
                params.gravity = Gravity.TOP or Gravity.START
                btn.layoutParams = params
            }
        }
        applyAvailability()
    }

    fun toggle() {
        if (!isAvailable()) return
        if (isExpanded) collapse() else expand()
    }

    private fun expand() {
        if (!isAvailable()) return
        isExpanded = true
        mainBtn.animate().cancel()
        mainBtn.animate().rotation(45f).setDuration(200).start()

        val radius = dp(60).toFloat()
        val angles = listOf(-90.0, -45.0, 0.0)
        for (i in subBtns.indices) {
            val angle = Math.toRadians(angles[i])
            val dx = (radius * kotlin.math.cos(angle)).toFloat()
            val dy = (radius * kotlin.math.sin(angle)).toFloat()
            val btn = subBtns[i]
            btn.animate().cancel()
            btn.visibility = View.VISIBLE
            btn.isClickable = true
            btn.isEnabled = true
            btn.alpha = 0f
            btn.scaleX = 0f
            btn.scaleY = 0f
            btn.animate()
                .translationX(dx).translationY(dy)
                .scaleX(1f).scaleY(1f).alpha(1f)
                .setDuration(300 + i * 50L)
                .setInterpolator(OvershootInterpolator(2.0f))
                .start()
        }
    }

    private fun collapse() {
        if (!isExpanded) return
        isExpanded = false
        mainBtn.animate().cancel()
        mainBtn.animate().rotation(0f).setDuration(200).start()
        subBtns.forEach { btn ->
            btn.isClickable = false
            btn.isEnabled = false
            btn.animate().cancel()
            btn.animate()
                .translationX(0f).translationY(0f)
                .scaleX(0f).scaleY(0f).alpha(0f)
                .setDuration(200)
                .withEndAction { if (!isExpanded) btn.visibility = View.GONE }
                .start()
        }
    }

    private fun collapseImmediately() {
        isExpanded = false
        mainBtn.animate().cancel()
        mainBtn.rotation = 0f
        subBtns.forEach { btn ->
            btn.animate().cancel()
            btn.visibility = View.GONE
            btn.isClickable = false
            btn.isEnabled = false
            btn.alpha = 0f
            btn.scaleX = 0f
            btn.scaleY = 0f
            btn.translationX = 0f
            btn.translationY = 0f
        }
    }

    fun destroy() {
        anchorView?.viewTreeObserver?.let { observer ->
            preDrawListener?.let { if (observer.isAlive) observer.removeOnPreDrawListener(it) }
        }
        preDrawListener = null
        anchorView = null
        isExpanded = false
        mainBtn.animate().cancel()
        mainBtn.setOnClickListener(null)
        subBtns.forEach {
            it.animate().cancel()
            it.setOnClickListener(null)
            rootLayout.removeView(it)
        }
        rootLayout.removeView(mainBtn)
        subBtns.clear()
    }
}

import android.graphics.Canvas

import android.graphics.PixelFormat

// Vòng tiến trình chạy dọc theo viền ô địa chỉ hình viên thuốc
class PillProgress(color: Int, private val strokePx: Float) : Drawable() {
    private val paint = Paint(Paint.ANTI_ALIAS_FLAG)
    private val path = Path()
    private val seg = Path()
    private val measure = PathMeasure()
    private var length = 0f

    var progress: Float = 0f
        set(v) {
            field = v.coerceIn(0f, 1f)
            invalidateSelf()
        }

    init {
        paint.color = color
        paint.style = Paint.Style.STROKE
        paint.strokeWidth = strokePx
        paint.strokeCap = Paint.Cap.ROUND
    }

    override fun onBoundsChange(bounds: Rect) {
        path.reset()
        val half = strokePx / 2f
        val rect = RectF(
            bounds.left + half,
            bounds.top + half,
            bounds.right - half,
            bounds.bottom - half
        )
        val r = minOf(rect.width(), rect.height()) / 2f
        path.addRoundRect(rect, r, r, Path.Direction.CW)
        measure.setPath(path, false)
        length = measure.length
    }

    override fun draw(canvas: Canvas) {
        if (progress <= 0f || progress >= 1f || length <= 0f) return
        seg.reset()
        measure.getSegment(0f, length * progress, seg, true)
        canvas.drawPath(seg, paint)
    }

    override fun setAlpha(alpha: Int) {
        paint.alpha = alpha
    }

    override fun setColorFilter(colorFilter: ColorFilter?) {
        paint.colorFilter = colorFilter
    }

    @Suppress("OVERRIDE_DEPRECATION")
    override fun getOpacity(): Int = PixelFormat.TRANSLUCENT
}

data class Entry(val title: String, val url: String, val time: Long)

class Store(context: Context) {
    private val sp = context.getSharedPreferences("edge", Context.MODE_PRIVATE)

    var homeUrl: String
        get() = sp.getString("home", "https://www.google.com") ?: "https://www.google.com"
        set(v) { sp.edit().putString("home", v).apply() }

    // 0 = Google, 1 = Bing, 2 = DuckDuckGo
    var searchEngine: Int
        get() = sp.getInt("engine", 0)
        set(v) { sp.edit().putInt("engine", v).apply() }

    var desktopMode: Boolean
        get() = sp.getBoolean("desktop", false)
        set(v) { sp.edit().putBoolean("desktop", v).apply() }

    var edgeMode: Boolean
        get() = sp.getBoolean("edge", true)
        set(v) { sp.edit().putBoolean("edge", v).apply() }

    var autoHideBar: Boolean
        get() = sp.getBoolean("autohide", true)
        set(v) { sp.edit().putBoolean("autohide", v).apply() }

    fun list(key: String): MutableList<Entry> {
        val arr = try {
            JSONArray(sp.getString(key, "[]") ?: "[]")
        } catch (e: Exception) {
            JSONArray()
        }
        val out = mutableListOf<Entry>()
        for (i in 0 until arr.length()) {
            val o = arr.getJSONObject(i)
            out.add(Entry(o.optString("t"), o.optString("u"), o.optLong("d")))
        }
        return out
    }

    private fun save(key: String, items: List<Entry>) {
        val arr = JSONArray()
        for (e in items) {
            arr.put(JSONObject().put("t", e.title).put("u", e.url).put("d", e.time))
        }
        sp.edit().putString(key, arr.toString()).apply()
    }

    fun addHistory(title: String, url: String) {
        if (url.isBlank() || url.startsWith("about:")) return
        val l = list("history")
        val entry = Entry(title.ifBlank { url }, url, System.currentTimeMillis())
        if (l.isNotEmpty() && l[0].url == url) {
            l[0] = entry
        } else {
            l.add(0, entry)
        }
        while (l.size > 300) l.removeAt(l.size - 1)
        save("history", l)
    }

    fun remove(key: String, url: String) {
        val l = list(key)
        l.removeAll { it.url == url }
        save(key, l)
    }

    fun clearHistory() {
        save("history", emptyList())
    }

    fun isBookmarked(url: String): Boolean = list("bookmarks").any { it.url == url }

    // Trả về true nếu sau khi bấm thì trang đã được thêm vào dấu trang
    fun toggleBookmark(title: String, url: String): Boolean {
        val l = list("bookmarks")
        val idx = l.indexOfFirst { it.url == url }
        return if (idx >= 0) {
            l.removeAt(idx)
            save("bookmarks", l)
            false
        } else {
            l.add(0, Entry(title.ifBlank { url }, url, System.currentTimeMillis()))
            save("bookmarks", l)
            true
        }
    }
}

package com.example.webedge

import android.graphics.Bitmap
import android.os.Bundle
import android.webkit.WebView

/**
 * Quản lý đa tab cho Summit Browser.
 * Dùng chung 1 WebView để nhẹ và tiết kiệm RAM.
 */
class TabManager {
    data class Tab(
        val id: Long = System.nanoTime(),
        var url: String = "",
        var title: String = "",
        var isActive: Boolean = false,

        var bundle: Bundle? = null,

        var thumbnail: Bitmap? = null
    )

    private val tabs = mutableListOf<Tab>()
    private var listener: (() -> Unit)? = null

    init {
        tabs.add(Tab(isActive = true))
    }

    fun setOnChangeListener(l: () -> Unit) { listener = l }
    fun getAll(): List<Tab> = tabs.toList()
    fun getActive(): Tab? = tabs.find { it.isActive }
    fun count(): Int = tabs.size

    fun addNew(url: String = "", title: String = "Tab mới"): Tab {
        tabs.forEach { it.isActive = false }
        val newTab = Tab(url = url, title = title, isActive = true)
        tabs.add(newTab)
        listener?.invoke()
        return newTab
    }

    fun switchTo(id: Long): Tab? {
        tabs.forEach { it.isActive = (it.id == id) }
        listener?.invoke()
        return tabs.find { it.id == id }
    }

    fun close(id: Long) {
        if (tabs.size <= 1) return
        val wasActive = tabs.find { it.id == id }?.isActive ?: false
        val idx = tabs.indexOfFirst { it.id == id }
        if (idx < 0) return
        tabs.removeAt(idx)
        if (wasActive) {
            val newActive = if (idx < tabs.size) tabs[idx] else tabs.last()
            newActive.isActive = true
        }
        listener?.invoke()
    }


    fun saveState(web: WebView) {
        val active = getActive() ?: return
        val oldUrl = active.url
        active.url = web.url ?: active.url
        active.title = web.title ?: active.title
        val urlChanged = oldUrl != active.url
        // Luu Bundle de khoi phuc lich su + vi tri cuon
        try {
            val b = Bundle()
            web.saveState(b)
            active.bundle = b
        } catch (_: Exception) {}
        // Chup thumbnail nho gon (chi khi URL thay doi)
        if (urlChanged) try {
            val w = web.width
            val h = web.height
            if (w > 0 && h > 0) {
                val scale = (180f * web.resources.displayMetrics.density) / w.toFloat()
                val tw = (w * scale).toInt().coerceAtLeast(1)
                val th = (h * scale).toInt().coerceAtLeast(1)
                val full = Bitmap.createBitmap(w, h, Bitmap.Config.ARGB_8888)
                val canvas = android.graphics.Canvas(full)
                web.draw(canvas)
                val scaled = Bitmap.createScaledBitmap(full, tw, th, true)
                if (scaled !== full) full.recycle()
                active.thumbnail?.recycle()
                active.thumbnail = scaled
            }
        } catch (_: Exception) {}
    }

    fun updateActive(url: String, title: String) {
        getActive()?.let {
            it.url = url
            it.title = title
        }
        listener?.invoke()
    }
}

package com.example.webedge

import android.graphics.Color

/**
 * Theme colors cho Summit Browser.
 * Tập trung các màu hardcode để dễ bảo trì và thay đổi.
 */
object Theme {
    // Màu accent chính (xanh dương iOS)
    val ACCENT_BLUE = Color.rgb(10, 132, 255)  // #0A84FF
    
    // Màu cảnh báo / xóa (đỏ)
    val DANGER_RED = Color.rgb(255, 69, 58)  // #FF453A
    
    // Màu chữ phụ (sub color)
    fun subColor(isNight: Boolean): Int =
        if (isNight) Color.argb(150, 255, 255, 255)
        else Color.argb(140, 0, 0, 0)
    
    // Màu gap (đường kẻ ngăn cách trong menu)
    fun gapColor(isNight: Boolean): Int =
        if (isNight) Color.argb(45, 255, 255, 255)
        else Color.argb(30, 0, 0, 0)
    
    // Màu fade ở đầu trang
    fun fadeStart(isNight: Boolean): Int =
        if (isNight) Color.argb(200, 0, 0, 0)
        else Color.argb(200, 255, 255, 255)
    
    // Màu fade ở cuối trang (trong suốt)
    fun fadeEnd(isNight: Boolean): Int =
        if (isNight) Color.argb(0, 0, 0, 0)
        else Color.argb(0, 255, 255, 255)
    
    // Màu nền trang
    fun pageColor(isNight: Boolean): Int =
        if (isNight) Color.rgb(18, 18, 18)
        else Color.WHITE
}
